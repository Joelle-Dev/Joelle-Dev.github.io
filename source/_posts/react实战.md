环境搭配(mac)

```
sudo npm install -g create-react-app
create-react-app my_cnode

//redux
npm i redux -S

//react-redux 整合
npm i react-redux -S

//中间件-异步请求
npm i redux-thunk -S

//路由
npm i react-router-dom -S

//axios
npm i axios -S

//antd
npm i antd -S
npm i react-app-rewired -D
//按需加载
npm i babel-plugin-import -D
```
